### FIND APPLICATIONS ###

find_program(
    YOSYS_EXECUTABLE
    NAMES yosys
    PATHS "${OSS_CAD_ROOT}/bin"
    NO_DEFAULT_PATH
    REQUIRED
)

find_program(
    NEXTPNR_ECP5_EXECUTABLE
    NAMES nextpnr-ecp5
    PATHS "${OSS_CAD_ROOT}/bin"
    NO_DEFAULT_PATH
    REQUIRED
)

find_program(
    ECPPACK_EXECUTABLE
    NAMES ecppack
    PATHS "${OSS_CAD_ROOT}/bin"
    NO_DEFAULT_PATH
    REQUIRED
)

find_program(
    IVERILOG_EXECUTABLE
    NAMES iverilog
    PATHS "${OSS_CAD_ROOT}/bin"
    NO_DEFAULT_PATH
    REQUIRED
)

find_program(
    VVP_EXECUTABLE
    NAMES vvp
    PATHS "${OSS_CAD_ROOT}/bin"
    NO_DEFAULT_PATH
    REQUIRED
)

find_program(
    OPENFPGALOADER_EXECUTABLE
    NAMES openFPGALoader
)


### LOCAL VARIABLES ###

set(FPGA_TOP digir_bpm_top)
set(FPGA_RTL_DIR ${CMAKE_CURRENT_SOURCE_DIR}/rtl)
set(FPGA_TB_DIR ${CMAKE_CURRENT_SOURCE_DIR}/tb)

file(GLOB_RECURSE FPGA_RTL
    CONFIGURE_DEPENDS
    "${FPGA_RTL_DIR}/*.sv"
)

file(GLOB_RECURSE FPGA_TESTBENCHES
    CONFIGURE_DEPENDS
    "${FPGA_TB_DIR}/*_tb.sv"
)

set(FPGA_CONSTRAINTS ${CMAKE_CURRENT_SOURCE_DIR}/constraints/ecp5_evn.lpf)
set(FPGA_BUILD_DIR "${CMAKE_CURRENT_BINARY_DIR}")
set(FPGA_SIM_BUILD_DIR "${FPGA_BUILD_DIR}/sim")
set(FPGA_YOSYS_SCRIPT "${FPGA_BUILD_DIR}/${FPGA_TOP}.ys")
set(FPGA_JSON "${FPGA_BUILD_DIR}/${FPGA_TOP}.json")
set(FPGA_CONFIG "${FPGA_BUILD_DIR}/${FPGA_TOP}.config")
set(FPGA_BITSTREAM "${FPGA_BUILD_DIR}/${FPGA_TOP}.bit")
set(FPGA_TIMING_REPORT "${FPGA_BUILD_DIR}/${FPGA_TOP}_timing.json")

file(MAKE_DIRECTORY "${FPGA_BUID_DIR}")
file(MAKE_DIRECTORY "${FPGA_SIM_BUILD_DIR}")


### GENERATE YOSYS SYNTHESIS SCRIPT ###

file (WRITE "${FPGA_YOSYS_SCRIPT}"
    "# Generated by CMake\n\n"
)

foreach(RTL_FILE IN LISTS FPGA_RTL)
    file(APPEND "${FPGA_YOSYS_SCRIPT}"
        "read_verilog -sv ${RTL_FILE}\n"
    )
endforeach()

file(APPEND "${FPGA_YOSYS_SCRIPT}"
    "\n"
    "hierarchy -check -top ${FPGA_TOP}\n"
    "synth_ecp5 -top ${FPGA_TOP} -json ${FPGA_JSON}\n"
    "stat\n"
)


### PERFORM SYNTHESIS ###

add_custom_command(
    OUTPUT
        "${FPGA_JSON}"

    COMMAND
        "${CMAKE_COMMAND}" -E make_directory "${FPGA_BUILD_DIR}"

    COMMAND
        "${YOSYS_EXECUTABLE}"
        -s "${FPGA_YOSYS_SCRIPT}"

    DEPENDS
        "${FPGA_RTL}"
        "${FPGA_YOSYS_SCRIPT}"

    COMMENT
        "Synthesizing ${FPGA_TOP} for ECP5"

    VERBATIM
)

add_custom_target(
    fpga_synth
    DEPENDS "${FPGA_JSON}"
)


### PLACE AND ROUTE ###

add_custom_command(
    OUTPUT
        "${FPGA_CONFIG}"

    BYPRODUCTS
        "${FPGA_TIMING_REPORT}"

    COMMAND
        "${NEXTPNR_ECP5_EXECUTABLE}"
        "--${FPGA_DEVICE}"
        --package "${FPGA_PACKAGE}"
        --speed "${FPGA_SPEED_GRADE}"
        --json "${FPGA_JSON}"
        --top "${FPGA_TOP}"
        --lpf "${FPGA_CONSTRAINTS}"
        --lpf-allow-unconstrained        # TODO: REMOVE FOR REAL HARDWARE
        --textcfg "${FPGA_CONFIG}"
        --report "${FPGA_TIMING_REPORT}"

    DEPENDS
        "${FPGA_JSON}"
        "${FPGA_CONSTRAINTS}"

    COMMENT
        "Placing and routing ${FPGA_TOP}"

    VERBATIM
)

add_custom_target(
    fpga_pnr
    DEPENDS "${FPGA_CONFIG}"
)


### PACK BITSTREAM ###

add_custom_command(
    OUTPUT
        "${FPGA_BITSTREAM}"

    COMMAND
        "${ECPPACK_EXECUTABLE}"
        "${FPGA_CONFIG}"
        "${FPGA_BITSTREAM}"

    DEPENDS
        "${FPGA_CONFIG}"

    COMMENT
        "Packing ${FPGA_TOP} ECP5 bitstream"

    VERBATIM
)

add_custom_target(
    fpga_bitstream
    ALL
    DEPENDS "${FPGA_BITSTREAM}"
)


### PROGRAM FPGA VOLATILE SRAM ###

if(OPENFPGALOADER_EXECUTABLE)
    add_custom_target(
        fpga_program

        COMMAND
            "${OPENFPGALOADER_EXECUTABLE}"
            --board ecp5_evn
            "${FPGA_BITSTREAM}"

        DEPENDS
            "${FPGA_BITSTREAM}"

        COMMENT
            "Writing FPGA configuration flash"

        USES_TERMINAL
        VERBATIM
    )

    add_custom_target(
        fpga_flash

        COMMAND
            "${OPENFPGALOADER_EXECUTABLE}"
            --board ecp5_evn
            --write-flash
            "${FPGA_BITSTREAM}"

        DEPENDS
            "${FPGA_BITSTREAM}"

        COMMENT
            "Writing FPGA configuration flash"

        USES_TERMINAL_VERBATIM
    )
else()
    message(STATUS
        "openFPGALoader not found; "
        "fpga_program and fpga_flash will not be available"
    )
endif()


### BUILD TESTBENCHES ###

add_custom_target(fpga_test)

foreach(TESTBENCH_SOURCE IN LISTS FPGA_TESTBENCHES)

    # Extract the module name identifying both the rtl and the testbench
    get_filename_component(
        TESTBENCH_MODULE
        "${TESTBENCH_SOURCE}"
        NAME_WE
    )
    string(
        REGEX REPLACE
        "_tb$"
        ""
        MODULE_NAME
        "${TESTBENCH_MODULE}"
    )

    # Compiled Icarus simulation output
    set(SIM_EXECUTABLE "${FPGA_SIM_BUILD_DIR}/${TESTBENCH_MODULE}.vvp")

    # Create CMake target names for each testbench. Format: fpga_compile_module_tb fpga_test_module
    set(COMPILE_TARGET "fpga_compile_${MODULE_NAME}_tb")
    set(TEST_TARGET "fpga_test_${MODULE_NAME}")


    ### COMPILE SIMULATION ###

    add_custom_command(
        OUTPUT "${SIM_EXECUTABLE}"

        COMMAND
            "${IVERILOG_EXECUTABLE}"
            -g2012
            -Wall
            -s "${TESTBENCH_MODULE}"
            -o "${SIM_EXECUTABLE}"
            ${FPGA_RTL}
            "${TESTBENCH_SOURCE}"

        DEPENDS
            ${FPGA_RTL}
            "${TESTBENCH_SOURCE}"

        COMMENT
            "Compiling FPGA testbench ${TESTBENCH_MODULE}"

        VERBATIM
    )

    ### RUN SIMULATION ###
    add_custom_target(
        "${TEST_TARGET}"

        COMMAND
            "${VVP_EXECUTABLE}"
            "${SIM_EXECUTABLE}"

        DEPENDS
            "${SIM_EXECUTABLE}"

        COMMENT
            "Running FPGA testbench ${TESTBENCH_MODULE}"

        USES_TERMINAL
        VERBATIM
    )

    ### ADD TEST TO AGGREGATE TARGET ###

    add_dependencies(
        fpga_test
        "${TEST_TARGET}"
    )

endforeach()
